NS.books.bank (assets/js/books-bank.js; loads after books.js):
  addAccount(ws, offeringId, { name, last4 }) → account; setMap(ws, offeringId, accountId, map)
  importLines(ws, offeringId, accountId, parsed, { name, sha256, fileId, by, at, allowOverlap }) → { added, skipped, overlap }    a line whose id is stored for that account is skipped; a line whose date, cents and text equal a stored line's under another id counts as overlap and is not added unless allowOverlap; the import record is stored only when added > 0
  undecided(ws, offeringId, accountId) → lines with neither match nor entry
  candidates(ws, offeringId, lineId) → [ { entry, line, date, cents, memo, partner, rank } ]   journal lines on the account's code, same cents (sign included), not matched by another stored line, and dated within 3 business days of the line (NS.ops.businessDays) or, for a run entry's line carrying a partner (a check), from its date to 180 days after; ranked first where the line's text holds the investor id or the offering's wire reference (NS.ops.wire(...).reference words), then by date distance
  match(ws, offeringId, lineId, { entry, line }) → line           refuses a journal line already matched, a different amount, or a line already decided
  autoMatch(ws, offeringId, accountId) → n                         matches each undecided line that has exactly one candidate, where that candidate is a candidate of no other undecided line
  categorize(ws, offeringId, lineId, { kind, account, holding, partner, costCents }, { by, at }) → entry    posts src 'bank:<lineId>' dated the line's date with cash code C and amount c (cents, money in positive):
    kind 'account': c > 0 → [C +c][account -c]; c < 0 → [account +|c|][C -|c|]; holding and partner go on the account line when given; account must be in the chart and not C
    kind 'purchase' (c < 0, holding required): [1100 +|c| '' holding][C -|c|]
    kind 'split' (lines required: [ { account, cents, holding, partner } ], each cents > 0, their sum = |c|, no account the cash code): one line per item on the side opposite the cash, then the cash line (a mortgage payment split between interest and principal)
    kind 'sale' (c > 0, holding and costCents required, costCents <= the holding's 1100 balance): u = round(1150 balance of the holding x costCents / 1100 balance of the holding); [C +c][1100 -costCents '' holding][6000 -(c - costCents) '' holding][6100 +u '' holding][1150 -u '' holding], zero lines left out
  uncategorize(ws, offeringId, lineId, { by, at }) → reverses the bank entry through NS.books.reverse (refused in a closed period) and clears line.entry
  reconcile(ws, offeringId, accountId, asOf, statementCents) → { statementCents, bookCents, outstanding: [ { entry, line, date, cents, memo } ], undecided: [ line ], differenceCents }   bookCents = balance of C at asOf; outstanding = journal lines on C dated <= asOf with no stored line dated <= asOf matched to them and no bank line that posted them; undecided = stored lines dated <= asOf with neither match nor entry; differenceCents = statementCents - bookCents + sum(outstanding cents) - sum(undecided cents)
  markReconciled(ws, offeringId, accountId, asOf, statementCents, { by, at }) → rec    refused unless differenceCents is 0
