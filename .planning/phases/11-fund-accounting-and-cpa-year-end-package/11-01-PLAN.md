NS.books (assets/js/books.js; pure; loads in the browser after audit.js and under Node; uses NS.audit.sha256 and NS.dates at call time; never writes the change log, never saves, never reads the engine):
  chart(o) → [ { code, name, type, line } ]                     the template for the offering
  accounts(ws, offeringId) → the template plus the fund's added accounts
  fund(ws, offeringId) → fund                                    created with start '' and empty lists when absent
  check(entry, accounts) → '' or one plain sentence saying what is wrong
  post(ws, offeringId, { date, memo, src, lines }, { by, at }) → entry      throws with check's sentence; throws 'closed' when date <= closedThrough; when a live entry with the same src exists it is returned and nothing is added
  reverse(ws, offeringId, entryId, { by, at, date, memo }) → the reversing entry (lines negated, rev set, src 'rev:<id>'); throws when already reversed or when either date is in a closed period; afterwards the reversed entry's src may be posted again
  live(fund, src) → the entry with that src that no reversal names, or null
  trialBalance(ws, offeringId, from, to) → [ { code, name, type, line, begin, debits, credits, end } ]   cents; debits and credits positive; end = begin + debits - credits; income, expense and gain accounts begin from 1 January of from's year and their earlier totals are added to 3900's begin; rows only for accounts with a balance or activity, in code order
  entries(ws, offeringId, { from, to, account, partner, holding, src }) → [ { id, date, memo, src, code, cents, partner, holding } ]   one row per line, by date then id; src matches as a prefix
  balance(ws, offeringId, code or [codes], asOf, { partner, holding }) → cents
  series(ws, offeringId, [codes], from, to) → [ { month: 'YYYY-MM', cents } ]   net activity per month, every month present
  tbText(ws, offeringId, through) → 'code|end' lines joined by '\n' over trialBalance(start of books, through)
  close(ws, offeringId, through, { by, at }) → closed record; through must be a month-end after closedThrough
  reopen(ws, offeringId, { by, at, why }) → the record reopened; why must not be empty; closedThrough becomes the previous through or ''
  closedThrough(ws, offeringId) → 'YYYY-MM-DD' or ''
  status(ws, offeringId) → { start, closedThrough, entries, exceptions }
  addAccount(ws, offeringId, { code, name, type, line }) → account (four digits, unused, a known type)
  addHolding(ws, offeringId, { name, kind, level, address, placed, landCents, buildingCents }) → holding (id 'h-' + base-36 count); setHolding(ws, offeringId, id, fields) → holding
